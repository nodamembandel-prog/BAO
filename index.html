// ========================================== // 1. SISTEM AUTO-UPDATE // ========================================== const APP_VERSION_CODE = 1; const VERSION_URL = "https://USERNAME_GITHUB_ANDA.github.io/NAMA_REPO_ANDA/version.json"; // PASTIKAN URL INI BENAR async function checkForUpdates() { try { let response = await fetch(VERSION_URL + "?t=" + new Date().getTime()); let data = await response.json(); if (data.versionCode > APP_VERSION_CODE) { let wantUpdate = confirm("Update Tersedia (" + data.versionName + ")!\n\n" + data.updateMessage + "\n\nUnduh sekarang?"); if (wantUpdate) { window.location.href = data.downloadUrl; } } } catch (e) { console.log("Gagal mengecek pembaruan: ", e); } } setTimeout(checkForUpdates, 3000); // ========================================== // 2. PANEL ADMIN TERSEMBUNYI & KEAMANAN PIN // ========================================== const PIN_ADMIN = "123456"; // GANTI ANGKA INI DENGAN PIN RAHASIA ANDA let logoClicks = 0; let clickTimer; document.getElementById("logoBao").addEventListener("click", function() { logoClicks++; clearTimeout(clickTimer); clickTimer = setTimeout(() => { logoClicks = 0; }, 2000); if (logoClicks >= 5) { bukaPanelAdmin(); logoClicks = 0; } }); function bukaPanelAdmin() { let inputPin = prompt("🔒 Masukkan PIN Admin:"); if (inputPin !== PIN_ADMIN) { alert("❌ Akses Ditolak! PIN salah."); return; } let currentAdUnitId = localStorage.getItem("admob_ad_unit_id") || ""; let newAdUnitId = prompt("=== PANEL ADMIN BAO ===\n\nMasukkan AdMob AD UNIT ID (Banner) khusus untuk HP ini:", currentAdUnitId); if (newAdUnitId !== null) { localStorage.setItem("admob_ad_unit_id", newAdUnitId); alert("✅ ID Iklan Lokal berhasil disimpan! Silakan muat ulang (restart) aplikasi."); } } // ========================================== // 3. INISIALISASI IKLAN (LOKAL & GLOBAL SERVER) // ========================================== async function initAds() { // 1. Cek iklan lokal (Prioritas untuk HP Anda sendiri) let finalAdUnitId = localStorage.getItem("admob_ad_unit_id"); // 2. Jika lokal kosong (Konsumen Play Store), ambil ID resmi dari Github secara diam-diam if (!finalAdUnitId) { try { let response = await fetch(VERSION_URL + "?t=" + new Date().getTime()); let data = await response.json(); if (data.adMobBannerId) { finalAdUnitId = data.adMobBannerId; } } catch (e) { console.log("Gagal mengambil data iklan dari server."); } } // Batalkan jika ID iklan tidak ada sama sekali if (!finalAdUnitId) return; try { const { AdMob } = window.Capacitor.Plugins; await AdMob.initialize({ requestTrackingAuthorization: true }); await AdMob.showBanner({ adId: finalAdUnitId, adSize: 'BANNER', position: 'BOTTOM_CENTER', margin: 0, isTesting: true // Ubah menjadi false sebelum publikasi final ke Play Store }); } catch (e) { console.log('Sistem iklan hanya berjalan di dalam APK Android.'); } } setTimeout(initAds, 2000);
